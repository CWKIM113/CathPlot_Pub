gnu_generator.py에 적용할 개선 포인트
현재 _nice_axis_range 메서드는 DAT 파일을 직접 열어서 min/max를 다시 계산하고 있습니다. 
하지만 이미 data_parser.py를 통해 JSON에 각 변수의 min_val과 max_val이 캐싱되어 있으므로, 
이 값을 가져와서 스케일을 계산하면 속도가 훨씬 빨라지고 정확해집니다.

이를 위해 CurveDefinition 또는 PlotVariable의 통계 정보를 활용하도록 
_nice_axis_range 부분을 다음과 같이 보완하실 수 있습니다.

1. _nice_axis_range 수정 방향 예시
def _nice_axis_range(self, plotdef, axis):
        """저장된 min/max 정보를 활용하여 가독성 좋은 (min, max, major-step) 반환"""
        values = []
        for curve in plotdef.curves:
            if getattr(curve, "axis", "y1") != axis:
                continue
            
            # 만약 curve에 min_val, max_val 속성이 있다면 우선 활용
            if hasattr(curve, "min_val") and hasattr(curve, "max_val"):
                values.extend([curve.min_val, curve.max_val])
            else:
                # Fallback: 기존처럼 직접 파일 읽기
                column = curve.output_no
                try:
                    with open(curve.dat_file, "r", encoding="utf-8", errors="ignore") as data:
                        for row in data:
                            fields = row.split()
                            if len(fields) <= column: continue
                            try:
                                val = float(fields[column].replace("D", "E").replace("d", "e"))
                                if math.isfinite(val): values.append(val)
                            except ValueError: continue
                except OSError: continue

        if not values:
            return None

        data_min, data_max = min(values), max(values)
        span = data_max - data_min
        if span == 0:
            span = max(abs(data_min) * 0.1, 1.0)
            data_min -= span / 2
            data_max += span / 2
        else:
            data_min -= span * 0.05
            data_max += span * 0.05
            span = data_max - data_min

        raw_step = span / 4.0
        magnitude = 10 ** math.floor(math.log10(raw_step))
        normalized = raw_step / magnitude
        for candidate in (1, 2, 2.5, 5, 10):
            if normalized <= candidate:
                step = candidate * magnitude
                break

        axis_min = math.floor(data_min / step) * step
        axis_max = math.ceil(data_max / step) * step
        if axis_min == axis_max:
            axis_max += step
        return axis_min, axis_max, step


data_parser.py 내 _parse_generic 수정본
수정 핵심 포인트
헤더 분리 방식 개선: re.split(r'[,\s]+', header_line.strip())를 사용하여 쉼표와 공백이 혼재된 헤더
(Time  pressure flowrate temp1...)를 정확하게 단어 단위로 쪼개어 raw_headers 리스트로 확보합니다.

샘플 데이터 연동: 주신 샘플 파일은 데이터 컬럼이 9개(Time 포함)인데 헤더에 명시된 이름은 
6개(Time, pressure, flowrate, temp1, temp2, viodfr)입니다. 
이 경우 범위를 벗어나는 나머지 뒤쪽 컬럼들(var_6, var_7, var_8)은 자동으로 var_# 이름이 부여되도록 
안전장치를 유지했습니다.

캐시 초기화: 이미 잘못 파싱되어 생성된 기존 sample.json 파일이 있다면 지우고
(CWD 폴더 내 해당 .json 삭제) 다시 실행하셔야 새로운 헤더가 정상적으로 반영됩니다.

def _parse_generic(self, lines, path):
        # 1. 데이터 시작 위치 및 헤더 줄 탐색
        data_start_line = 0
        header_line = ""
        for i, line in enumerate(lines):
            if self._is_numeric_row(line):
                data_start_line = i
                # 데이터 줄 바로 위에서 헤더 탐색 (구분선 '---' 등 제외)
                if i > 0 and not self._is_numeric_row(lines[i-1]) and "---" not in lines[i-1]:
                    header_line = lines[i-1]
                break
        
        # 2. 숫자 데이터 행 추출 및 컬럼별 min/max 계산
        data_rows = []
        for line in lines[data_start_line:]:
            if self._is_numeric_row(line):
                # 쉼표나 공백으로 구분된 데이터를 안전하게 float로 변환
                vals = [float(x.replace(',', '')) for x in re.split(r'[,\s]+', line.strip()) if x]
                if vals:
                    data_rows.append(vals)

        num_cols = len(data_rows[0]) if data_rows else 0
        
        col_stats = {}
        for c in range(num_cols):
            col_vals = [row[c] for row in data_rows if len(row) > c]
            if col_vals:
                col_stats[c] = (min(col_vals), max(col_vals))
            else:
                col_stats[c] = (0.0, 1.0)

        # 3. 헤더 파싱 (공백 및 쉼표 처리 강화)
        raw_headers = []
        if header_line:
            # 쉼표(,)와 공백(\s+)을 모두 분리 기준으로 사용하고 빈 문자열 제거
            raw_headers = [h.strip() for h in re.split(r'[,\s]+', header_line.strip()) if h]

        # 4. PlotVariable 생성 (헤더 우선 사용, 부족하면 var_# 적용)
        variables = []
        for i in range(num_cols):
            if i == 0: 
                continue # 첫 번째 컬럼(Time)은 변수 목록에서 제외
            
            # 헤더 목록에 해당 인덱스의 이름이 존재하면 사용, 없거나 비어있으면 var_i 사용
            if i < len(raw_headers) and raw_headers[i]:
                name = raw_headers[i]
            else:
                name = f"var_{i}" 
            
            c_min, c_max = col_stats.get(i, (0.0, 1.0))

            variables.append(PlotVariable(
                output_no=i, 
                data_column=i + 1,
                signals=[SignalDefinition(name=name)],
                min_val=c_min,
                max_val=c_max
            ))
            
        return DatFile(
            filename=path.name,
            date_line="Generic Format",
            title_lines=[lines[0].strip() if lines else "Title"],
            format_line="Generic Table",
            n_outputs=num_cols - 1,
            variables=variables,
            data_start_line=data_start_line
        )
