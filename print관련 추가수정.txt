def create_multiplot_script(self, plots, outfile, fmt="pdf", layout_option="Auto", orientation="portrait", margins=(0.08, 0.94, 0.06, 0.94), spacing=(0.09, 0.10)):
        plot_dir = os.path.dirname(os.path.abspath(outfile))
        os.makedirs(plot_dir, exist_ok=True)

        nplot = len(plots)
        rows, cols = 1, 1
        
        if layout_option == "Auto":
            if nplot <= 1: rows, cols = 1, 1
            elif nplot == 2: rows, cols = (1, 2) if orientation == "landscape" else (2, 1)
            elif nplot <= 4: rows, cols = 2, 2
            else: rows, cols = 3, 2
        elif layout_option in ["1/p", "1"]: rows, cols = 1, 1
        elif "2/p" in layout_option: rows, cols = 2, 1
        elif "4/p" in layout_option: rows, cols = 2, 2
        elif "6/p" in layout_option: rows, cols = 3, 2

        w, h = self._A4_SIZES.get(orientation, self._A4_SIZES["portrait"])
        size_override = f"{w},{h}"

        lines = [
            "set encoding utf8",
            self._get_terminal_cmd(fmt, outfile, None, size_override),
            f"set output '{outfile.replace(chr(92), '/')}'"
        ]

        l, r, b, t = margins
        sx, sy = spacing

        if rows * cols > 1:
            lines.append(f"set multiplot layout {rows},{cols} margins {l},{r},{b},{t} spacing {sx},{sy}")
        else:
            lines.append(f"set multiplot layout {rows},{cols}")

        for i, plotdef in enumerate(plots):
            # 다중 플롯 시 앞선 그림의 잔상(축/margin)이 남아 겹치거나 밀리는 현상 방지
            if i > 0:
                lines.append("unset title")
                lines.append("set xtics auto")
                lines.append("set ytics auto")
                lines.append("set y2tics auto")
            lines.append("set lmargin")
            lines.append("set rmargin")
            lines.append("set tmargin")
            lines.append("set bmargin")
            self._build_plot_commands(plotdef, lines, is_multiplot=True)

        lines.append("unset multiplot")
        lines.append("set output")

        script_path = outfile.rsplit(".", 1)[0] + ".plt"
        with open(script_path, "w", encoding="utf-8") as f:
            f.write("\n".join(lines))

        try:
            gnuplot_exe = _resolve_gnuplot_exe()
            subprocess.run([gnuplot_exe, script_path], check=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        finally:
            if os.path.exists(script_path):
                os.remove(script_path)

        return outfile

def create_multiplot_script(self, plots, outfile, fmt="pdf", layout_option="Auto", orientation="portrait", margins=(0.08, 0.94, 0.06, 0.94), spacing=(0.09, 0.10), keep_script=False, frame_auto=True, frame_width="", frame_height="", frame_unit="in"):
        plot_dir = os.path.dirname(os.path.abspath(outfile))
        os.makedirs(plot_dir, exist_ok=True)

        nplot = len(plots)
        rows, cols = 1, 1

        if layout_option == "Auto":
            if nplot <= 1: rows, cols = 1, 1
            elif nplot == 2: rows, cols = (1, 2) if orientation == "landscape" else (2, 1)
            elif nplot <= 4: rows, cols = 2, 2
            else: rows, cols = 3, 2
        elif layout_option in ["1/p", "1"]: rows, cols = 1, 1
        elif "2/p" in layout_option: rows, cols = 2, 1
        elif "4/p" in layout_option: rows, cols = 2, 2
        elif "6/p" in layout_option: rows, cols = 3, 2

        if not frame_auto and frame_width and frame_height:
            if fmt.lower() in ("png", "jpg", "jpeg"):
                size_override = f"{frame_width},{frame_height}"
            else:
                size_override = f"{frame_width}{frame_unit},{frame_height}{frame_unit}"
        else:
            w, h = self._A4_SIZES.get(orientation, self._A4_SIZES["portrait"])
            size_override = f"{w},{h}"

        lines = [
            "set encoding utf8",
            self._get_terminal_cmd(fmt, outfile, None, size_override),
            f"set output '{outfile.replace(chr(92), '/')}'"
        ]

        # margins/spacing를 "set multiplot layout" 한 줄에 같이 주는 방식은
        # gnuplot 버전에 따라 파싱이 안 될 수 있고, 그러면 layout 자체가
        # 무효화되어 모든 subplot이 분할 없이 전체 화면에 겹쳐 그려진다.
        # 대신 각 칸의 위치를 여기서 직접 계산해서 매 subplot마다
        # lmargin/rmargin/tmargin/bmargin을 절대좌표(at screen)로 지정한다.
        # 이 방식은 gnuplot 버전과 무관하게 항상 동작하는 표준적인 방법이다.
        l, r, b, t = margins
        sx, sy = spacing

        total_w = r - l
        total_h = t - b
        cell_w = (total_w - sx * (cols - 1)) / cols
        cell_h = (total_h - sy * (rows - 1)) / rows

        lines.append(f"set multiplot layout {rows},{cols}")

        for i, plotdef in enumerate(plots):
            row_idx = i // cols
            col_idx = i % cols

            cell_left = l + col_idx * (cell_w + sx)
            cell_right = cell_left + cell_w
            cell_top = t - row_idx * (cell_h + sy)
            cell_bottom = cell_top - cell_h

            if i > 0:
                lines.append("unset title")
                lines.append("set xtics auto")
                lines.append("set ytics auto")
                lines.append("set y2tics auto")

            lines.append(f"set lmargin at screen {cell_left:.4f}")
            lines.append(f"set rmargin at screen {cell_right:.4f}")
            lines.append(f"set tmargin at screen {cell_top:.4f}")
            lines.append(f"set bmargin at screen {cell_bottom:.4f}")

            self._build_plot_commands(plotdef, lines, is_multiplot=True)

        lines.append("unset multiplot")
        lines.append("set output")

        script_path = outfile.rsplit(".", 1)[0] + ".plt"
        with open(script_path, "w", encoding="utf-8") as f:
            f.write("\n".join(lines))

        gnuplot_exe = _resolve_gnuplot_exe()
        result = subprocess.run(
            [gnuplot_exe, script_path],
            capture_output=True, text=True
        )

        if result.returncode != 0:
            raise RuntimeError(
                f"gnuplot 실행 실패 (스크립트 보존됨: {script_path})\n"
                f"{result.stderr.strip()}"
            )

        if not keep_script and os.path.exists(script_path):
            os.remove(script_path)

        return outfile
