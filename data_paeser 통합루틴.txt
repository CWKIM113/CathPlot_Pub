def parse(self, filename):
        path = Path(filename)
        with open(filename, 'r', encoding='utf-8', errors='ignore') as f:
            lines = [line.strip() for line in f.readlines()]

        if not lines:
            raise ValueError(f"File is empty: {filename}")

        # 포맷 판별 로직 (CATHENA 포맷 특유의 시그니처가 있는지 확인)
        # 예: CATHENA 포맷은 특정 줄에 포맷 라인이나 날짜 라인 구조가 명확함
        if self._is_cathena_format(lines):
            return self._parse_cathena(lines, path)
        else:
            return self._parse_generic(lines, path)

    def _is_cathena_format(self, lines):
        # CATHENA 파일인지 판별하는 조건 검사
        # (예: 특정 줄에 CATHENA 고유 키워드가 있거나, 표준 헤더 구조를 만족하는 경우 True 반환)
        if len(lines) > 5:
            # 예시 조건: 폼 라인이나 CATHENA 전용 키워드 감지
            for line in lines[:10]:
                if "CATHENA" in line.upper() or "DATAPLOT" in line.upper():
                    return True
        # 기본적으로는 명확한 CATHENA 형식이 아니면 Generic으로 처리
        return False
